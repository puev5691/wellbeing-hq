import copy, unittest
import runtime_integration_tests as rit
from runtime_integration import ContractCompilerFacade
from sandbox_adapter import ADAPTER_ID, EphemeralFileSandboxEffectAdapterR01, ObjectBoundCleanupR02
from sandbox_adapter_tests import bind, created, cstate, platform
from sandbox_runtime_integration import SandboxBoundEffectIntentEmitter, SandboxBoundPreEffectRevalidator, SandboxBoundEffectBoundaryVerifier

def combined():
    env,p,resolution,actor,task,frontier,pdep = rit.bundle()
    compiled=ContractCompilerFacade(rit.FakeCore()).compile(env,resolution,actor)
    b=bind()
    frontier=dict(frontier)
    frontier.update({str(k):str(v) for k,v in b["supporting_evidence_versions"].items()})
    authority=rit.adapter_authority()
    authority["adapter_class"]=ADAPTER_ID
    intent=SandboxBoundEffectIntentEmitter().emit(
        env,resolution,compiled,actor,
        sandbox_effect_binding=b,
        required_outcome_evidence_mode="EXACT_RESULT",
    )
    admission=SandboxBoundPreEffectRevalidator().revalidate(
        intent,compiled,actor,
        current_evidence_versions=frontier,
        expected_evidence_versions={},
        adapter_authority=authority,
        prior_effect_state="NONE",
        current_trust_policy_dependency=pdep,
        current_task_execution_binding=task,
        current_sandbox_effect_binding=copy.deepcopy(b),
    )
    invocation=rit.invocation(actor,task,frontier,pdep,authority)
    invocation["sandbox_effect_binding"]=copy.deepcopy(b)
    return env,resolution,actor,task,frontier,pdep,compiled,b,authority,intent,admission,invocation

    def test_01_happy_path_binding_admission_and_boundary(self):
        *_, b, authority, intent, admission, invocation = combined()[7:]
